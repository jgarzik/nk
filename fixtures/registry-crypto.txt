# Registry content for Microsoft's kernel cryptography and security
# drivers, laid over the SYSTEM hive with registry-overlay.txt in the
# boots that ask for it (a scenario's `registry = ["crypto"]`;
# tools/mkesp.py).
#
# cng.sys and ksecdd.sys are boot drivers on Windows: the reference
# installation's driver list shows both "Boot", "Running", "Kernel"
# (sc/driverquery). ksecdd.sys opens \Device\CNG in its DriverEntry, so
# CNG starts first. ksecdd.sys also registers performance counters
# (PcwRegister) and fails its start if that fails; the kernel's Pcw
# routines are pcw.sys's, which the reference installation loads at boot
# too.
#
# Every value is that of the SYSTEM hive of the Windows 10 22H2 WinPE
# image (boot.wim, image 1, Windows\System32\config\SYSTEM), the
# destination's own registry: pcw in group System Reserved, the first of
# ServiceGroupOrder; CNG in group Core; KSecDD in group Base; both of the
# latter critical (ErrorControl 3).

key ControlSet001\Services\pcw
value ImagePath expand_sz System32\drivers\pcw.sys
value Type dword 1
value Start dword 0
value ErrorControl dword 1
value Group sz "System Reserved"
value DisplayName sz "Performance Counters for Windows Driver"

key ControlSet001\Services\CNG
value ImagePath expand_sz System32\Drivers\cng.sys
value Type dword 1
value Start dword 0
value ErrorControl dword 3
value Group sz Core
value Tag dword 4

key ControlSet001\Services\KSecDD
value ImagePath expand_sz System32\Drivers\ksecdd.sys
value Type dword 1
value Start dword 0
value ErrorControl dword 3
value Group sz Base
value Tag dword 1
